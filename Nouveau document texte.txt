  <button onclick="poserQuestion()">Envoyer</button>
  <div class="reponse" id="reponse">La réponse apparaîtra ici...</div>

  <script>
    async function poserQuestion() {
      const question = document.getElementById("question").value;
      const reponseDiv = document.getElementById("reponse");
      reponseDiv.innerText = "Je réfléchis... 🤔";

      const apiKey = "ta_clé_API_ici"; // ← remplace par ta vraie clé ici

      const data = {
        model: "gpt-3.5-turbo",
        messages: [{ role: "user", content: question }]
      };

      try {
        const response = await fetch("https://api.openai.com/v1/chat/completions", {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${apiKey}`
          },
          body: JSON.stringify(data)
        });

        const result = await response.json();
        reponseDiv.innerText = result.choices[0].message.content;
      } catch (error) {
        reponseDiv.innerText = "Erreur : " + error.message;
      }
    }
  </script>
